## استفاده

```php
use Tavaan\Engine\Tenancy\TenantManager;
use Tavaan\Engine\Tenancy\Tenant;
use Tavaan\Engine\Tenancy\TenantContext;
use Tavaan\Engine\Tenancy\InMemoryTenantRepository;
use Tavaan\Engine\Tenancy\TenantResolveChain;
use Tavaan\Engine\Tenancy\Resolvers\DomainResolver;
use Tavaan\Engine\Tenancy\Resolvers\HeaderResolver;
use Tavaan\Engine\Tenancy\Resolvers\PathResolver;

// Setup with Repository
$repo = new InMemoryTenantRepository();
$repo->create(['identifier' => 'acme', 'plan' => 'pro', 'domains' => ['acme.example.com']]);
$m = new TenantManager([], $repo);

// Resolve tenant from request
$chain = new TenantResolveChain([
    new DomainResolver('example.com'),
    new HeaderResolver('X-Tenant-ID'),
    new PathResolver(1),
]);
$identifier = $chain->resolve(['host' => 'acme.example.com', 'headers' => [], 'path' => '/']);

// Set current tenant
$tenant = $m->findByIdentifier($identifier);
$m->setCurrent($tenant);

// Access current
$m->current();      // ?Tenant
$m->id();           // 1
$m->identifier();   // 'acme'
$m->quota('orders_per_day');  // 1000

// Run within tenant context
$m->run($anotherTenant, function () {
    // اجرا در زمینۀ tenant دیگر
});

// Cross-tenant (with permission)
$m->withoutCurrent(function () {
    // اجرای بدون tenant scope
});

// Find by domain
$m->findByDomain('acme.example.com');

// Create new
$new = $m->create(['identifier' => 'new-tenant', 'plan' => 'free']);
```
